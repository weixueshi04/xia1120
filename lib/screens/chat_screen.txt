// 鑷畾涔夎湣鏌撹儗鏅粯鍒剁被
class BatikBackgroundPainter extends CustomPainter {
  @override
  void paint(Canvas canvas, Size size) {
    final paint = Paint()
      ..color = MiaoTheme.indigoDye.withAlpha((0.05 * 255).round())
      ..style = PaintingStyle.fill;

    // 鑳屾櫙濉厖
    canvas.drawRect(Rect.fromLTWH(0, 0, size.width, size.height), paint);

    // 缁樺埗绠€鍗曠殑铚℃煋椋庢牸鍥炬
    final patternPaint = Paint()
      ..color = MiaoTheme.indigoDye.withAlpha((0.1 * 255).round())
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1.5;

    // 缁樺埗缃戞牸鐘跺浘妗堟ā鎷熻湣鏌撶汗鐞?
    const gridSize = 30.0;
    for (double x = 0; x <= size.width; x += gridSize) {
      canvas.drawLine(Offset(x, 0), Offset(x, size.height), patternPaint);
    }
    for (double y = 0; y <= size.height; y += gridSize) {
      canvas.drawLine(Offset(0, y), Offset(size.width, y), patternPaint);
    }

    // 缁樺埗涓€浜涘渾鐐硅楗?
    final dotPaint = Paint()
      ..color = MiaoTheme.scholarGold.withAlpha((0.1 * 255).round())
      ..style = PaintingStyle.fill;

    for (double x = 0; x < size.width; x += 60) {
      for (double y = 0; y < size.height; y += 60) {
        canvas.drawCircle(Offset(x + 30, y + 30), 3, dotPaint);
      }
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

                    setState(() {
                      _isRecording = true;
                    });
                    
                    ScaffoldMessenger.of(context).showSnackBar(
                      const SnackBar(
                        content: Text('馃帳 姝ｅ湪褰曢煶...'),
                        duration: Duration(seconds: 10),

                    setState(() {
                      _isRecording = false;

                      const SnackBar(
                        content: Text('馃攧 姝ｅ湪璇嗗埆...'),
                        duration: Duration(seconds: 3),

                    );

                      _textController.text = text;
                      _handleSend();
                    } else {
                      ScaffoldMessenger.of(context).clearSnackBars();
                      ScaffoldMessenger.of(context).showSnackBar(
                        const SnackBar(
                          content: Text('鉂?鏈瘑鍒埌鍐呭'),
                          duration: Duration(seconds: 2),

                      );

                  }

                  width: 48,
                  height: 48,
                  decoration: BoxDecoration(
                    color: _isRecording 
                        ? MiaoTheme.mapleRed.withAlpha((0.2 * 255).round())
                        : MiaoTheme.indigoDye.withAlpha((0.1 * 255).round()),
                    shape: BoxShape.circle,
                  ),
                  child: Icon(
                    _isRecording ? Icons.mic : Icons.mic_none,
                    color: _isRecording ? MiaoTheme.mapleRed : MiaoTheme.indigoDye,
                  ),
                ),

  void _openGame(GameDefinition game) async {
    String? suggestedQuestion;

    if (game.id == 'book_filing') {
      suggestedQuestion = await Navigator.of(context).push<String>(
        MaterialPageRoute(builder: (_) => const GameBookFilingScreen()),
      );
    } else if (game.id == 'farm_management') {
      suggestedQuestion = await Navigator.of(context).push<String>(
        MaterialPageRoute(builder: (_) => const GameFarmManagementScreen()),
      );
    }

    if (!mounted) return;
    if (suggestedQuestion != null && suggestedQuestion.isNotEmpty) {
      setState(() {
        _textController.text = suggestedQuestion!;
      });
    }
  }

  /// 鏄犲皠鍒嗙被鍒版秷鎭被鍒?
  MessageCategory? _mapCategoryToMessageCategory(String category) {
    switch (category) {
      case '闈為仐':
        return MessageCategory.culture;
      case '鍐滀笟':
        return MessageCategory.agriculture;
      case '瀛︿範':
        return MessageCategory.education;
      case '鏁呬簨':
        return MessageCategory.story;
      default:
        return null;
    }
  }

  /// 浠庢煡璇㈡枃鏈娴嬪垎绫?
  MessageCategory? _detectCategoryFromQuery(String query) {
    final lowerQuery = query.toLowerCase();
    
    // 闈為仐鏂囧寲鍏抽敭璇?
    if (lowerQuery.contains('铚℃煋') || lowerQuery.contains('闈為仐') || 
        lowerQuery.contains('鍒虹唬') || lowerQuery.contains('閾堕グ') ||
        lowerQuery.contains('鑺︾瑱') || lowerQuery.contains('澶ф瓕')) {
      return MessageCategory.culture;
    }
    
    // 鍐滀笟鐭ヨ瘑鍏抽敭璇?
    if (lowerQuery.contains('钃濊帗') || lowerQuery.contains('鍐滀笟') || 
        lowerQuery.contains('绉嶆') || lowerQuery.contains('楹诲皬鑾?) ||
        lowerQuery.contains('鍦熷￥') || lowerQuery.contains('鐥呰櫕瀹?)) {
      return MessageCategory.agriculture;
    }
    
    // 鏁欒偛杈呭鍏抽敭璇?
    if (lowerQuery.contains('瀛︿範') || lowerQuery.contains('浣滀笟') || 
        lowerQuery.contains('鑰冭瘯') || lowerQuery.contains('杈呭') ||
        lowerQuery.contains('璇讳功') || lowerQuery.contains('鎴愮哗')) {
      return MessageCategory.education;
    }
    
    // 鏁呬簨鏂囧寲鍏抽敭璇?
    if (lowerQuery.contains('鏁呬簨') || lowerQuery.contains('鑺傛棩') || 
        lowerQuery.contains('鑻楁棌') || lowerQuery.contains('浼犺') ||
        lowerQuery.contains('鍘嗗彶') || lowerQuery.contains('鏅偣')) {
      return MessageCategory.story;
    }
    
    return null;
  }

  Future<void> _handleSend() async {

    _textController.clear();

    final userMessage = Message.user(text);
    setState(() {
      _messages.add(userMessage);
      _isProcessing = true;
      _avatarState = AvatarState.thinking;
    });

    _scrollToBottom();

    String? answer;
    MessageCategory? category;

    // 1. 鍏堝皾璇?RAG 妫€绱?
    final resp = await _ragService.search(text);
    if (resp != null) {
      answer = resp.content;
      category = _detectCategoryFromQuery(text);

    // 2. 濡傛灉鏈湴娌℃湁绛旀锛岃皟鐢ㄧ櫨搴gent

      answer = await _baiduService.chat(text);
      
      if (answer != null) {
        category = _detectCategoryFromQuery(text);

    setState(() {
